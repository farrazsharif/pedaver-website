# Copy this file into the top of the `out/` folder as `.htaccess` before zipping
# and uploading to cPanel. It sets a custom 404 page and asset caching.

# Custom 404 page
ErrorDocument 404 /404.html

# Farmer Stories was merged into the Knowledge & Testimonials page
RedirectMatch 301 ^/farmer-stories/?$ /videos/

# Farmer Advisory is hidden until the four PQNK books are published — its
# page route is de-registered (src/app/_advisory) so `out/advisory/` now
# only holds the mobile app's manifest.json. Send the bare /advisory page
# URL to Knowledge Exchange (its eventual umbrella) instead of letting the
# directory 403 or list. The anchored pattern leaves /advisory/manifest.json
# untouched for the app. Remove this line when the page is re-activated.
RedirectMatch 301 ^/advisory/?$ /field-evidence/

# Certification was renamed to Validation (PQNK validates practice, it doesn't
# certify organic status) — redirect at the server level so search engines and
# old links get a real 301, not just the page's client-side JS redirect
RedirectMatch 301 ^/certification/?$ /validation/

# Sample Page was a leftover from the pre-Next.js site (an old founder bio
# page). It no longer exists on the server at all, but Google's index still
# has it — redirect it to the current Founder page so that stale link/index
# entry resolves correctly instead of 404ing.
RedirectMatch 301 ^/sample-page/?$ /founder/

# Another pre-Next.js WordPress leftover — its entire body was a one-paragraph
# definition of what PQNK stands for ("PQNK ... means sustainable natural
# farming system"), confirmed via Wayback Machine snapshots (the page no
# longer exists on the server, but still ranks live for "Pedaver PQNK"
# searches and 404s). /science/ is the current site's dedicated "what PQNK is"
# hub and the closest canonical successor to that content.
RedirectMatch 301 ^/pqnk-paedar-qudratti-nizam-e-kashatqari/?$ /science/

# Cache static assets aggressively (they are content-hashed by Next)
<IfModule mod_expires.c>
  ExpiresActive On
  ExpiresByType text/css "access plus 1 year"
  ExpiresByType application/javascript "access plus 1 year"
  ExpiresByType image/png "access plus 6 months"
  ExpiresByType image/jpeg "access plus 6 months"
  ExpiresByType image/svg+xml "access plus 6 months"
  ExpiresByType image/x-icon "access plus 6 months"
  ExpiresByType text/html "access plus 0 seconds"
</IfModule>

# Gzip/text compression
<IfModule mod_deflate.c>
  AddOutputFilterByType DEFLATE text/html text/css application/javascript application/json image/svg+xml
</IfModule>

# Canonical host — www.pedaver.com is not the canonical domain; send it
# straight to the apex over HTTPS in a single redirect (checked first so an
# http:// www. request doesn't need two hops through the HTTPS-force rule
# below).
<IfModule mod_rewrite.c>
  RewriteEngine On
  RewriteCond %{HTTP_HOST} ^www\.pedaver\.com$ [NC]
  RewriteRule ^ https://pedaver.com%{REQUEST_URI} [L,R=301]
</IfModule>

# Force HTTPS — send all http:// visitors to the secure https:// version
<IfModule mod_rewrite.c>
  RewriteEngine On
  RewriteCond %{HTTPS} off
  RewriteRule ^ https://%{HTTP_HOST}%{REQUEST_URI} [L,R=301]
</IfModule>
